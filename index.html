Scraper Python Locale per Borsa Italiana (MOT)

Usa requests.Session() e regex per estrarre l'elenco completo in JSON/CSV.

Dipendenze: pip install requests beautifulsoup4 
import json
import re
import time
from bs4 import BeautifulSoup
import requests

BASE_URL = "https://www.borsaitaliana.it/borsa/obbligazioni/mot/btp/lista.html?page={}"

session = requests.Session()
session.headers.update({
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
        "(KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36"
    ),
    "Accept-Language": "it-IT,it;q=0.9,en-US;q=0.8,en;q=0.7",
    "Referer": "https://www.borsaitaliana.it/borsa/obbligazioni/mot/btp/lista.html",
})

def parse_it_float(raw_val):
    if not raw_val:
        return None
    cleaned = re.sub(r"[^\d,.-]", "", str(raw_val)).replace(".", "").replace(",", ".")
    try:
        return float(cleaned)
    except ValueError:
        return None

def extract_bonds_from_html(html_text):
    soup = BeautifulSoup(html_text, "html.parser")
    rows = soup.select("table tr")
    extracted = []

    for tr in rows:
        row_text = " ".join(tr.get_text(" ", strip=True).split())
        isin_match = re.search(r"\b(IT[0-9A-Z]{10})\b", row_text)
        if not isin_match:
            continue

        isin = isin_match.group(1)
        cells = [ " ".join(td.get_text(" ", strip=True).split()) for td in tr.find_all(["td", "th"]) ]

        # Estrazione prezzo 'Ultimo'
        ultimo_match = re.search(r"Ultimo[:\s]*([\d,]+)", row_text, re.IGNORECASE)
        price = parse_it_float(ultimo_match.group(1)) if ultimo_match else None
        if price is None:
            for c in cells[1:]:
                num = parse_it_float(c)
                if num is not None and 30.0 <= num <= 160.0:
                    price = num
                    break

        # Scarta titoli illiquidi / strip senza quotazione valida
        if price is None or price <= 0:
            continue

        # Estrazione descrizione e scadenza
        mat_match = re.search(r"\b(\d{2}[/.-]\d{2}[/.-]\d{4})\b", row_text)
        maturity = mat_match.group(1).replace("-", "/").replace(".", "/") if mat_match else ""

        name = cells[1] if len(cells) > 1 and cells[1] else isin

        # Cedola periodica -> annua (semestrale * 2, trimestrale * 4 per BTP Valore / BTP Più)
        cedola_match = re.search(r"Cedola[:\s]*([\d,]+)", row_text, re.IGNORECASE)
        periodic_coupon = parse_it_float(cedola_match.group(1)) if cedola_match else 0.0
        multiplier = 4 if re.search(r"\b(valore|piu|più)\b", name, re.IGNORECASE) else 2
        annual_coupon = round((periodic_coupon or 0.0) * multiplier, 4)

        if annual_coupon == 0.0:
            pct_match = re.search(r"(\d+[.,]\d+|\d+)\s*%", name)
            if pct_match:
                annual_coupon = parse_it_float(pct_match.group(1)) or 0.0

        extracted.append({
            "isin": isin,
            "name": name,
            "price": price,
            "coupon": annual_coupon,
            "maturity": maturity,
        })

    return extracted

all_bonds = {}
for page in range(1, 16):
    url = BASE_URL.format(page)
    try:
        resp = session.get(url, timeout=15)
        resp.raise_for_status()
    except Exception as exc:
        print(f"[!] Errore pagina {page}: {exc}")
        break

    page_bonds = extract_bonds_from_html(resp.text)
    if not page_bonds:
        break

    new_count = 0
    for b in page_bonds:
        if b["isin"] not in all_bonds:
            all_bonds[b["isin"]] = b
            new_count += 1

    print(f"Pagina {page}: trovati {len(page_bonds)} titoli ({new_count} nuovi)")
    if new_count == 0:
        break
    time.sleep(0.5)

result = list(all_bonds.values())
with open("btp_mot_borsa_italiana.json", "w", encoding="utf-8") as f:
    json.dump(result, f, ensure_ascii=False, indent=2)

print(f"Completato: {len(result)} BTP salvati in btp_mot_borsa_italiana.json")
